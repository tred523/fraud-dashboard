In client/src/pages/Accounts.jsx find where last_seen is displayed and apply the same date fix:
const fmt = (ts) => {
  if (!ts) return '—';
  const n = Number(ts);
  if (isNaN(n)) return ts;
  const d = new Date(n > 9999999999 ? n : n * 1000);
  return isNaN(d.getTime()) ? String(ts) : d.toLocaleString('en-GB');
};
